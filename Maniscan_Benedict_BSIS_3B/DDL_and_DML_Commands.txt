<--- DATA DEFINITION LANGUAGE (DDL)--->


CREATE DATABASE bsis3b;

CREATE TABLE employee_information(
	id INT PRIMARY KEY,
	EmployeeID VARCHAR(50),
	FirstName VARCHAR(50),
	LastName VARCHAR(50),
	Gender VARCHAR(50),
	BirthDate DATE(50),
	ContactNo VARCHAR(50),
	PositionID VARCHAR(50),
	DepartmentID VARCHAR(50)
);

CREATE TABLE Department (
	id INT PRIMARY KEY,
	DepartmentID VARCHAR(50),
	DepartmentName VARCHAR(50),
	Location VARCHAR(50)
);

CREATE TABLE Position_Table (
	id INT PRIMARY KEY,
	PositionID VARCHAR(50),
	JobTitle VARCHAR(50),
	Salary INT(50),
	JobDescription VARCHAR(50)
);

CREATE TABLE Attendance_Table (
	id INT PRIMARY KEY,
	AttendanceID VARCHAR(50),
	EmployeeID VARCHAR (50),
	Date DATE,
	Timein TIME,
	TimeOut TIME,
	Status VARCHAR(50)
);

CREATE TABLE payroll_table (
	id INT PRIMARY KEY,
	PayrollID VARCHAR(50),
	EmployeeID VARCHAR (50),
	PayPertiod Date,
	BasicSalary INT(50),
	Deduction INT(50),
	NetSalary INT(50)
);
